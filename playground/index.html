piragi

zero-config RAG — try it interactively

pip install piragi
GitHub
1. Setup 
pip install piragi
2. Basic RAG 
from piragi import Ragi

kb = Ragi("./docs")
answer = kb.ask("How do I deploy?")
print(answer.text)
print(f"Citations: {len(answer.citations)}")
3. ONNX Speedup 
from piragi import Ragi

kb = Ragi("./docs", config={"embedding": {"backend": "onnx"}})
# 2-5x faster on CPU
4. Shared Embedder 
from piragi import Ragi, EmbeddingGenerator

embedder = EmbeddingGenerator(model="all-mpnet-base-v2")
kb_docs = Ragi("./docs", embedder=embedder)
kb_code = Ragi("./src", embedder=embedder)
# One model, multiple knowledge bases
5. Qdrant + TurboQuant 
from piragi.stores import QdrantStore

store = QdrantStore(
    url="http://localhost:6333",
    quantization="turboquant",
)
kb = Ragi("./docs", store=store)
6. Async 
from piragi import AsyncRagi

async_kb = AsyncRagi("./docs")
answer = await async_kb.ask("What's the API?")
